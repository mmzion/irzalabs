    <script>
        const firebaseConfig = {
            apiKey: "AIzaSyC8NcbOPzhbUnm9NxOZyzdEJSLk_LOGYcc",
            authDomain: "irza-labs.firebaseapp.com",
            databaseURL: "https://irza-labs-default-rtdb.firebaseio.com",
            projectId: "irza-labs",
            storageBucket: "irza-labs.firebasestorage.app",
            messagingSenderId: "269557073409",
            appId: "1:269557073409:web:4c2977681884d733f62200"
        };
        firebase.initializeApp(firebaseConfig);
        const db = firebase.database();

        // ============ GLOBAL STATE ============
        const rooms = ['room1', 'room2'];
        const roomData = {};
        rooms.forEach(r => roomData[r] = {
            power:0, status:"OFF", current:0, allocated:0,
            manual_cutoff:false, voltage:0, max_watt_limit:0,
            lastUpdate:0
        });
        let chart;

        // ============ INIT CHART ============
        function initChart() {
            const ctx = document.getElementById('liveChart').getContext('2d');
            chart = new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: ['Room 01', 'Room 02'],
                    datasets: [{
                        label: 'kWh Used',
                        data: [0, 0],
                        backgroundColor: ['rgba(59,130,246,0.6)', 'rgba(168,85,247,0.6)'],
                        borderColor: ['#3b82f6', '#a855f7'],
                        borderWidth: 1
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: { beginAtZero: true, grid: { color: 'rgba(255,255,255,0.05)' }, ticks: { color: '#94a3b8' } },
                        x: { ticks: { color: '#94a3b8' } }
                    },
                    plugins: { legend: { display: false } }
                }
            });
        }

        // ============ LOGIN ============
        function handleLogin() {
            const email = document.getElementById('loginEmail').value;
            const pass = document.getElementById('loginPassword').value;
            firebase.auth().signInWithEmailAndPassword(email, pass)
                .then(() => {
                    document.getElementById('loginScreen').style.display = 'none';
                    document.getElementById('adminDashboard').style.display = 'block';
                    initChart();
                    rooms.forEach(r => listenToRoom(r));
                    listenToLogs();
                    setInterval(checkOnlineStatus, 2000);
                })
                .catch(() => {
                    document.getElementById('loginError').innerText = 'Invalid credentials!';
                    document.getElementById('loginError').style.display = 'block';
                });
        }

        function handleLogout() {
            firebase.auth().signOut().then(() => location.reload());
        }

        // ============ LISTEN TO ROOM DATA ============
        function listenToRoom(roomName) {
            db.ref('rooms/' + roomName).on('value', snap => {
                const data = snap.val();
                if (!data) return;
                const rd = roomData[roomName];
                rd.power = data.power || 0;
                rd.status = data.status || "OFF";
                rd.allocated = data.allocated_units || 0;
                rd.current = data.current_units || 0;
                rd.voltage = data.voltage || 0;
                rd.manual_cutoff = data.manual_cutoff === true;
                rd.max_watt_limit = data.max_watt_limit || 0;   // FIX: store watt limit
                rd.lastUpdate = Date.now();

                updateRoomUI(roomName);
                updateChart();
                updateGlobalStats();
            });
        }

        function updateRoomUI(roomName) {
            const rd = roomData[roomName];
            const card = document.getElementById(`card-${roomName}`);
            if (!card) return;

            // Online dot
            const dot = card.querySelector('.online-dot');
            if (dot) {
                const online = (Date.now() - rd.lastUpdate) < 5000;
                dot.className = 'online-dot' + (online ? '' : ' offline');
            }

            // Badge & status
            const badge = card.querySelector('.badge');
            const statusText = rd.status.includes("ON") ? "ON" : "OFF";
            badge.innerText = statusText;
            badge.className = 'badge ' + (statusText === 'ON' ? 'on' : 'off');

            // Data fields
            card.querySelector('.curr-val').innerText = rd.current.toFixed(2) + " kWh";
            card.querySelector('.alloc-val').innerText = rd.allocated.toFixed(2) + " kWh";
            card.querySelector('.power-val').innerText = rd.power.toFixed(1) + " W";
            // FIX: use rd.max_watt_limit instead of data.max_watt_limit
            card.querySelector('.watt-limit-val').innerText = (rd.max_watt_limit || 0) + " W";
            card.querySelector('.volt-val').innerText = rd.voltage.toFixed(1) + " V";

            // Cutoff button
            const cutBtn = card.querySelector('.btn-cutoff');
            if (rd.manual_cutoff) {
                cutBtn.innerHTML = '<i class="fa-solid fa-rotate-left"></i> Restore Power';
                cutBtn.className = 'btn-cutoff restore';
                cutBtn.onclick = () => toggleCutoff(roomName, false);
                card.classList.add('alert');
            } else {
                cutBtn.innerHTML = '<i class="fa-solid fa-power-off"></i> Shutdown';
                cutBtn.className = 'btn-cutoff';
                cutBtn.onclick = () => toggleCutoff(roomName, true);
                card.classList.remove('alert');
            }
        }

        function checkOnlineStatus() {
            rooms.forEach(r => {
                const rd = roomData[r];
                const dot = document.querySelector(`#card-${r} .online-dot`);
                if (dot) {
                    const online = (Date.now() - rd.lastUpdate) < 5000;
                    dot.className = 'online-dot' + (online ? '' : ' offline');
                }
            });
        }

        function updateChart() {
            chart.data.datasets[0].data = [roomData.room1.current, roomData.room2.current];
            chart.update();
        }

        function updateGlobalStats() {
            let totalPower = roomData.room1.power + roomData.room2.power;
            let activeCount = (roomData.room1.status.includes("ON") ? 1 : 0) + (roomData.room2.status.includes("ON") ? 1 : 0);
            document.getElementById('activeRooms').innerText = activeCount;
            document.getElementById('totalPower').innerText = totalPower.toFixed(1) + " W";
        }

        // ============ CONTROL FUNCTIONS ============
        function updateLimit(roomName) {
            const input = document.getElementById(`input-${roomName}`);
            const val = parseFloat(input.value);
            if (!isNaN(val) && val >= 0) {
                db.ref('rooms/' + roomName).update({ allocated_units: val });
                writeLog(roomName, "kWh Set", val + " kWh");
                input.value = '';
            }
        }

        function quickAdd(roomName, amount) {
            const current = roomData[roomName].allocated || 0;
            db.ref('rooms/' + roomName).update({ allocated_units: current + amount });
            writeLog(roomName, "+" + amount + " kWh", (current + amount) + " kWh");
        }

        function updateWattLimit(roomName) {
            const input = document.getElementById(`watt-input-${roomName}`);
            const val = parseFloat(input.value);
            if (!isNaN(val) && val >= 0) {
                db.ref('rooms/' + roomName).update({ max_watt_limit: val });
                writeLog(roomName, "Watt Limit", val + " W");
                input.value = '';
            }
        }

        function toggleCutoff(roomName, status) {
            db.ref('rooms/' + roomName).update({ manual_cutoff: status });
            writeLog(roomName, status ? "Shutdown" : "Restore", "manual");
        }

        function masterSystemReset() {
            if (confirm("Reset all rooms & logs?")) {
                rooms.forEach(r => db.ref('rooms/' + r).update({ current_units:0, manual_cutoff:false, allocated_units:0, max_watt_limit:0 }));
                db.ref('allocation_logs').remove();
                alert("System reset.");
            }
        }

        // ============ LOGGING ============
        function writeLog(room, type, val) {
            const now = new Date();
            const timeStr = now.toLocaleDateString() + ' ' + now.toLocaleTimeString();
            db.ref('allocation_logs').push({
                timestamp: firebase.database.ServerValue.TIMESTAMP,
                time: timeStr,
                room: room,
                type: type,
                units: val
            });
        }

        function listenToLogs() {
            db.ref('allocation_logs').orderByChild('timestamp').limitToLast(10).on('value', snap => {
                const tbody = document.getElementById('logBody');
                tbody.innerHTML = '';
                let count = 0;
                snap.forEach(child => {
                    const log = child.val();
                    count++;
                    const row = `<tr>
                        <td>${log.time}</td>
                        <td style="font-weight:600;color:var(--primary);">${log.room.toUpperCase()}</td>
                        <td>${log.type}</td>
                        <td>${log.units}</td>
                    </tr>`;
                    tbody.insertAdjacentHTML('afterbegin', row);
                });
                document.getElementById('totalLogs').innerText = count;
            });
        }

        // ============ BUILD ROOM CARDS (dynamic) ============
        function buildRoomCard(roomName) {
            const displayName = roomName === 'room1' ? 'Room 01' : 'Room 02';
            return `
                <div class="room-card" id="card-${roomName}">
                    <div class="room-header">
                        <div class="room-title">
                            <span class="online-dot offline"></span>
                            ${displayName}
                        </div>
                        <span class="badge off">OFF</span>
                    </div>
                    <div class="data-row"><span class="dim">Used</span> <span class="curr-val">0.00 kWh</span></div>
                    <div class="data-row"><span class="dim">Allocated</span> <span class="alloc-val">0.00 kWh</span></div>
                    <div class="data-row"><span class="dim">Power / Limit</span> <span><span class="power-val">0 W</span> / <span class="watt-limit-val">0 W</span></span></div>
                    <div class="data-row"><span class="dim">Voltage</span> <span class="volt-val">0 V</span></div>
                    <div class="controls">
                        <label>Energy Limit (kWh)</label>
                        <div class="input-row">
                            <input type="number" id="input-${roomName}" step="0.5" placeholder="kWh" style="flex:1;">
                            <button class="btn-action" onclick="updateLimit('${roomName}')"><i class="fa-solid fa-floppy-disk"></i></button>
                        </div>
                        <div style="display:flex; gap:6px; margin-bottom:12px;">
                            <button class="btn-quick" onclick="quickAdd('${roomName}',5)">+5 kWh</button>
                            <button class="btn-quick" onclick="quickAdd('${roomName}',10)">+10 kWh</button>
                        </div>
                        <label>Watt Limit (W)</label>
                        <div class="input-row">
                            <input type="number" id="watt-input-${roomName}" placeholder="Watts">
                            <button class="btn-action watt" onclick="updateWattLimit('${roomName}')"><i class="fa-solid fa-bolt"></i></button>
                        </div>
                        <button class="btn-cutoff" onclick="toggleCutoff('${roomName}', true)"><i class="fa-solid fa-power-off"></i> Shutdown</button>
                    </div>
                </div>
            `;
        }

        // Build room cards on load
        document.getElementById('roomsContainer').innerHTML = rooms.map(r => buildRoomCard(r)).join('');
    </script>
